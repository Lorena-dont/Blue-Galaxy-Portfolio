const canvas = document.getElementById('galaxy-canvas'); const ctx = canvas.getContext('2d'); let width, height; let particles = []; const particleCount = 2800; // Estado del ratón para interactividad let mouse = { x: 0, y: 0, targetX: 0, targetY: 0 }; function resize() { width = canvas.width = window.innerWidth; height = canvas.height = window.innerHeight; mouse.x = width / 2; mouse.y = height / 2; mouse.targetX = width / 2; mouse.targetY = height / 2; } window.addEventListener('resize', resize); window.addEventListener('mousemove', (e) => { mouse.targetX = e.clientX; mouse.targetY = e.clientY; }); window.addEventListener('touchmove', (e) => { if (e.touches.length > 0) { mouse.targetX = e.touches[0].clientX; mouse.targetY = e.touches[0].clientY; } }); resize(); class Particle { constructor() { this.init(); } init() { const distRatio = Math.pow(Math.random(), 2.5); this.radius = distRatio * (Math.max(width, height) * 0.65); const arms = 2; // Brazos espirales const armOffset = (Math.floor(Math.random() * arms) * (Math.PI * 2 / arms)); this.angle = armOffset + (this.radius * 0.0012); this.spreadX = (Math.random() - 0.5) * (this.radius * 0.22); this.spreadY = (Math.random() - 0.5) * (this.radius * 0.22); this.spinSpeed = 0.0002 + (1 / (this.radius + 60)) * 0.18; this.size = Math.random() * 1.8 + 0.3; const colors = ['#c084fc', '#d946ef', '#a855f7', '#38bdf8', '#ffffff', '#e879f9']; this.color = colors[Math.floor(Math.random() * colors.length)]; this.baseAlpha = Math.random() * 0.7 + 0.2; this.alpha = this.baseAlpha; this.twinkleSpeed = Math.random() * 0.02 + 0.005; } update() { this.angle += this.spinSpeed; // Efecto de parpadeo suave this.alpha += Math.sin(Date.now() * this.twinkleSpeed) * 0.008; this.alpha = Math.max(0.1, Math.min(0.9, this.alpha)); const rawX = Math.cos(this.angle) * this.radius + this.spreadX; const rawY = Math.sin(this.angle) * this.radius + this.spreadY; // Movimiento de cámara/paralaje con el cursor const offsetX = (mouse.x - width / 2) * (0.05 * (this.radius / (width * 0.5))); const offsetY = (mouse.y - height / 2) * (0.05 * (this.radius / (height * 0.5))); this.x = (width / 2) + rawX + offsetX; this.y = (height / 2) + rawY + offsetY; } draw() { ctx.save(); ctx.fillStyle = this.color; ctx.globalAlpha = this.alpha; if (this.size > 1.4) { ctx.shadowBlur = 8; ctx.shadowColor = this.color; } ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctx.fill(); ctx.restore(); } } function drawCoreGlow() { const coreX = (width / 2) + (mouse.x - width / 2) * 0.02; const coreY = (height / 2) + (mouse.y - height / 2) * 0.02; const gradient = ctx.createRadialGradient(coreX, coreY, 5, coreX, coreY, width * 0.35); gradient.addColorStop(0, 'rgba(217, 70, 239, 0.22)'); gradient.addColorStop(0.4, 'rgba(147, 51, 234, 0.08)'); gradient.addColorStop(1, 'rgba(3, 0, 8, 0)'); ctx.fillStyle = gradient; ctx.fillRect(0, 0, width, height); } function initGalaxy() { particles = []; for (let i = 0; i < particleCount; i++) { particles.push(new Particle()); } } function animate() { mouse.x += (mouse.targetX - mouse.x) * 0.05; mouse.y += (mouse.targetY - mouse.y) * 0.05; ctx.fillStyle = 'rgba(3, 0, 8, 0.25)'; ctx.fillRect(0, 0, width, height); drawCoreGlow(); particles.forEach(p => { p.update(); p.draw(); }); requestAnimationFrame(animate); } initGalaxy(); animate();
